/** * A3Sim v3.8-PRO: Simulation Execution Loop with TTT Evaluation Span */ function runSimulation(params) { const { distance, servBase, servSlope, neighBase, neighSlope, a3Offset, hysteresis, tttMs, rlfThreshold, fadeAmp } = params; let points = []; let servRSRP = []; let neighRSRP = []; let tttWindowArr = []; // Visual span for TTT evaluation duration let consecutiveCount = 0; let tttStartIndex = -1; let handoverIndex = -1; let rlfIndex = -1; let simulationStatus = "SUCCESS"; // Step conversion (assuming 1 step = 1 meter, time step = 10ms per meter) const timeStepMs = 10; const requiredSteps = Math.max(1, Math.round(tttMs / timeStepMs)); for (let i = 0; i <= distance; i++) { let x = i; points.push(x); // Calculate path loss with light fading let noiseS = (Math.sin(x * 0.3) * 0.5 + Math.cos(x * 0.1) * 0.5) * fadeAmp; let noiseN = (Math.cos(x * 0.25) * 0.5 + Math.sin(x * 0.15) * 0.5) * fadeAmp; let sVal = servBase - (servSlope * x) + noiseS; let nVal = neighBase + (neighSlope * x) + noiseN; servRSRP.push(sVal); neighRSRP.push(nVal); // Check for Radio Link Failure (RLF) condition if (sVal < rlfThreshold && rlfIndex === -1 && handoverIndex === -1) { rlfIndex = i; simulationStatus = "RLF_DROP"; break; // Call dropped } // Check if handover has already occurred if (handoverIndex !== -1) { tttWindowArr.push(null); continue; } // Event A3 Condition: Neighbor - Hysteresis > Serving + Offset let a3Condition = (nVal - hysteresis) > (sVal + a3Offset); if (a3Condition) { if (consecutiveCount === 0) { tttStartIndex = i; // Mark the beginning of the TTT evaluation span } consecutiveCount++; // Highlight active TTT window span on chart tttWindowArr.push(nVal); if (consecutiveCount >= requiredSteps) { handoverIndex = i; simulationStatus = "HO_SUCCESS"; break; } } else { // Leaky persistence: subtract 1 instead of instantly wiping out // so small fading dips don't unfairly penalize valid handovers consecutiveCount = Math.max(0, consecutiveCount - 1); if (consecutiveCount > 0) { tttWindowArr.push(nVal); } else { tttWindowArr.push(null); tttStartIndex = -1; } } } return { points, servRSRP, neighRSRP, tttWindowArr, handoverIndex, rlfIndex, simulationStatus }; }
